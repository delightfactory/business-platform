'use server';
import type {PreviewState,ConfirmState,ImportRow} from 'C:/Users/DELL/Documents/Codex/2026-10-07/business-platform-ux-auth-recovery/src/app/tenant/[tenantId]/attendance/import/actions';
export async function previewAttendanceCsv(previous:PreviewState,fd:FormData):Promise<PreviewState>{await new Promise(r=>setTimeout(r,1400));const file=fd.get('file') as File;const text=await file.text();const fail=text.includes('PREVIEW_FAIL');const statuses:ImportRow['status'][]=text.includes('REJECT_ONLY')?['rejected']:text.includes('DUPLICATE_ONLY')?['duplicate']:['ready','ready','ambiguous','unassigned','duplicate','rejected'];const rows=fail?[]:statuses.map((status,i)=>({source_line_hint:i+2,employee_code:'SYN-'+i,site_name:'فرع صناعي',happened_at:'2026-10-07T09:00:00+03:00',direction:'in',source_event_key:(text.includes('CONFIRM_FAIL')?'FAIL-':text.includes('AMBIGUOUS_RETRY')?'RETRY-':'TEST-')+i,status,errors:status==='rejected'?['رفض صناعي']:[],warnings:[],candidate_work_dates:status==='ambiguous'?['2026-10-07','2026-10-08']:[],work_date:status==='ready'?'2026-10-07':undefined}));return{tenantId:String(fd.get('tenantId')),rows,ready:rows.filter(r=>r.status==='ready').length,ambiguous:rows.filter(r=>r.status==='ambiguous').length,unassigned:rows.filter(r=>r.status==='unassigned').length,duplicate:rows.filter(r=>r.status==='duplicate').length,rejected:rows.filter(r=>r.status==='rejected').length,error:fail?'فشل معاينة صناعي':'',attempt:previous.attempt+1};}
export async function confirmAttendanceCsv(previous:ConfirmState,fd:FormData):Promise<ConfirmState>{await new Promise(r=>setTimeout(r,1400));const selected=fd.getAll('selectedRow').map(x=>JSON.parse(String(x)) as ImportRow);const fail=selected.some(x=>x.source_event_key.startsWith('FAIL-')) || (previous.state==='processed' && previous.ambiguous>0 && selected.some(x=>x.source_event_key.startsWith('RETRY-')));const rows=fail?[]:selected.map(x=>({...x,status:x.status==='ambiguous'&&(!fd.get('workDate_'+x.source_line_hint)||(x.source_event_key.startsWith('RETRY-')&&!previous.rows.some(y=>y.source_event_key===x.source_event_key&&y.status==='ambiguous')))?'ambiguous':x.status==='unassigned'?'unassigned':'accepted'} as ImportRow));return{state:fail?'failed':'processed',rows,accepted:rows.filter(r=>r.status==='accepted').length,ambiguous:rows.filter(r=>r.status==='ambiguous').length,unassigned:rows.filter(r=>r.status==='unassigned').length,duplicate:0,rejected:0,error:fail?'فشل تأكيد صناعي؛ المدخلات محفوظة':'',attempt:previous.attempt+1};}
